DATABASE SCHEMA EXPORT
========================


TABLE: approval_logs
--------------------
COLUMN                    | TYPE            | NULLABLE   | DEFAULT
----------------------------------------------------------------------
log_id                    | character varying | NO         | NULL
transaction_id            | character varying | YES        | NULL
approver_name             | character varying | YES        | NULL
approver_role             | character varying | YES        | NULL
timestamp                 | timestamp without time zone | YES        | NULL
action_taken              | character varying | YES        | NULL
amount_limit              | numeric         | YES        | NULL

TABLE: audit_results
--------------------
COLUMN                    | TYPE            | NULLABLE   | DEFAULT
----------------------------------------------------------------------
invoice_id                | character varying | NO         | NULL
risk_rating               | character varying | YES        | NULL
flags_detected            | jsonb           | YES        | NULL
explanation               | text            | YES        | NULL
recommended_action        | text            | YES        | NULL
citation                  | text            | YES        | NULL
policy_version_applied    | numeric         | YES        | NULL
session_id                | character varying | YES        | NULL

TABLE: company_policies
-----------------------
COLUMN                    | TYPE            | NULLABLE   | DEFAULT
----------------------------------------------------------------------
policy_id                 | integer         | NO         | nextval('company_policies_policy_id_seq'::regclass)
markdown_content          | text            | YES        | NULL
version_number            | numeric         | YES        | NULL
is_active                 | boolean         | YES        | NULL
created_at                | timestamp without time zone | YES        | CURRENT_TIMESTAMP

---------------------
COLUMN                    | TYPE            | NULLABLE   | DEFAULT

received_amount           | numeric         | YES        | NULL

  vendor_id         VARCHAR(20) PRIMARY KEY,
  vendor_name       VARCHAR(200) NOT NULL,
  official_gstin    VARCHAR(15),
  pan_number        VARCHAR(10),
  vendor_category   VARCHAR(50),             -- 'Logistics','IT Services','Raw Materials','MRO','Professional Services','Utilities'
  registered_address TEXT,                    -- backward compat with old schema
  city              VARCHAR(100),
  state             VARCHAR(100),
  country           VARCHAR(50) DEFAULT 'India',
  bank_account_number VARCHAR(50),
  bank_ifsc         VARCHAR(20),
  bank_verified     BOOLEAN DEFAULT FALSE,
  gstin_status      VARCHAR(20) DEFAULT 'Unverified',  -- 'Active','Mismatch','Suspended','Missing','Invalid','Unverified'
  kyc_status        VARCHAR(20) DEFAULT 'Pending',     -- 'Verified','Pending','Expired','Rejected'
  kyc_document_url  TEXT,
  kyc_verified_at   TIMESTAMP,
  risk_score_numeric INTEGER DEFAULT 0,
  risk_rating       VARCHAR(20),             -- backward compat with old schema
  shell_risk_score  INTEGER DEFAULT 0,       -- 0-100, for collusion detection

  shared_ip_address VARCHAR(45),             -- IP if shared with other vendors

  blacklist_status  BOOLEAN DEFAULT FALSE,   -- backward compat alias
  created_at        TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at        TIMESTAMP DEFAULT CURRENT_TIMESTAMP

-- Three-way match view (for Audit Workspace)
CREATE OR REPLACE VIEW v_three_way_match AS
SELECT
  i.invoice_id,
  i.vendor_id,
  vm.vendor_name,
  COALESCE(i.amount, i.invoice_amount) AS invoice_amount,
  COALESCE(po.amount, po.po_amount) AS po_amount,
  COALESCE(gr.amount, gr.received_amount) AS grn_amount,
  COALESCE(i.po_id, i.po_reference) AS po_reference,
  COALESCE(i.grn_id, i.grn_reference) AS grn_reference,
  CASE
    WHEN po.po_id IS NULL THEN 'Missing PO'
    WHEN gr.grn_id IS NULL THEN 'Missing GRN'
    WHEN ABS(COALESCE(i.amount, i.invoice_amount, 0) - COALESCE(po.amount, po.po_amount, 0))
         / NULLIF(COALESCE(po.amount, po.po_amount, 0), 0) > 0.05 THEN 'Amount Mismatch'
    WHEN ABS(COALESCE(po.amount, po.po_amount, 0) - COALESCE(gr.amount, gr.received_amount, 0))
         / NULLIF(COALESCE(po.amount, po.po_amount, 0), 0) > 0.05 THEN 'GRN Mismatch'
    ELSE 'Matched'
  END AS match_status
FROM invoices i
LEFT JOIN purchase_orders po ON COALESCE(i.po_id, i.po_reference) = po.po_id
LEFT JOIN goods_receipts gr ON COALESCE(i.grn_id, i.grn_reference) = gr.grn_id
LEFT JOIN vendor_master vm ON i.vendor_id = vm.vendor_id;

-- Daily audit trends (for Dashboard line chart)
CREATE OR REPLACE VIEW v_daily_audit_trends AS
SELECT
  DATE_TRUNC('day', created_at)::date AS audit_date,
  COUNT(*) AS total_audits,
  COUNT(*) FILTER (WHERE risk_rating = 'CRITICAL') AS critical,
  COUNT(*) FILTER (WHERE risk_rating = 'HIGH') AS high,
  COUNT(*) FILTER (WHERE risk_rating = 'MEDIUM') AS medium,
  COUNT(*) FILTER (WHERE risk_rating = 'LOW') AS low,
  COUNT(*) FILTER (WHERE workflow_status = 'Approved') AS approved,
  COUNT(*) FILTER (WHERE workflow_status = 'Pending Review') AS pending,
  ROUND(AVG(risk_score), 1) AS avg_risk_score
FROM audit_results
GROUP BY 1 ORDER BY 1;

-- Vendor risk heatmap (for Dashboard scatter plot + Vendor Intelligence)
CREATE OR REPLACE VIEW v_vendor_risk_heatmap AS
SELECT
  vm.vendor_id,
  vm.vendor_name,
  vm.vendor_category,
  vm.gstin_status,
  vm.kyc_status,
  vm.blacklisted,
  vm.shell_risk_score,
  vm.shared_director,
  vm.shared_ip_address,
  COUNT(ar.id) AS total_audits,
  COUNT(ar.id) FILTER (WHERE ar.risk_rating IN ('HIGH','CRITICAL')) AS high_risk_count,
  ROUND(AVG(ar.risk_score), 1) AS avg_risk_score,
  MAX(ar.created_at) AS last_audit_at,
  SUM(COALESCE(i.total_amount, i.amount, i.invoice_amount, 0)) AS total_invoice_value
FROM vendor_master vm
LEFT JOIN audit_results ar ON vm.vendor_id = ar.vendor_id
LEFT JOIN invoices i ON vm.vendor_id = i.vendor_id
GROUP BY vm.vendor_id, vm.vendor_name, vm.vendor_category, vm.gstin_status,
         vm.kyc_status, vm.blacklisted, vm.shell_risk_score, vm.shared_director, vm.shared_ip_address;

-- Rule violation feed (for analytics)
CREATE OR REPLACE VIEW v_rule_violation_feed AS
SELECT
  flag->>'flagName' AS rule_name,
  flag->>'severity' AS risk_level,
  flag->>'category' AS category,
  COUNT(*) AS trigger_count,
  COUNT(DISTINCT ar.vendor_id) AS vendors_affected,
  MAX(ar.created_at) AS last_triggered
FROM audit_results ar, jsonb_array_elements(ar.flags_detected) AS flag
WHERE ar.flags_detected IS NOT NULL AND ar.flags_detected != 'null'::jsonb
GROUP BY flag->>'flagName', flag->>'severity', flag->>'category'
ORDER BY trigger_count DESC;

-- ABC distribution (for Dashboard donut chart)
CREATE OR REPLACE VIEW v_abc_distribution AS
SELECT
  transaction_class,
  COUNT(*) AS count,
  SUM(COALESCE(i.amount, i.invoice_amount, 0)) AS total_value,
  ROUND(AVG(ar.risk_score), 1) AS avg_risk_score
FROM audit_results ar
LEFT JOIN invoices i ON ar.invoice_id = i.invoice_id
GROUP BY transaction_class;

-- Ingestion pipeline summary
CREATE OR REPLACE VIEW v_ingestion_summary AS
SELECT
  connector_type,
  COUNT(*) AS total_jobs,
  SUM(total_records) AS total_records,
  SUM(success_count) AS total_success,
  SUM(error_count) AS total_errors,
  SUM(duplicate_count) AS total_duplicates,
  ROUND(100.0 * SUM(success_count) / NULLIF(SUM(total_records), 0), 1) AS success_rate,
  MAX(uploaded_at) AS last_upload
FROM ingestion_jobs
GROUP BY connector_type;

-- Missed ITC view (for Tax Intelligence)
CREATE OR REPLACE VIEW v_missed_itc AS
SELECT
  i.invoice_id,
  i.vendor_id,
  vm.vendor_name,
  i.gstin AS invoice_gstin,
  vm.official_gstin,
  i.tax_amount AS missed_itc_amount,
  i.hsn_code,
  COALESCE(i.date, i.invoice_date::date) AS invoice_date
FROM invoices i
JOIN vendor_master vm ON i.vendor_id = vm.vendor_id
WHERE i.gstin IS NOT NULL
  AND vm.official_gstin IS NOT NULL
  AND i.gstin != vm.official_gstin;

-- Volume discount opportunities (for Tax Intelligence)
CREATE OR REPLACE VIEW v_volume_discounts AS
SELECT
  i.vendor_id,
  vm.vendor_name,
  vm.vendor_category,
  DATE_TRUNC('quarter', COALESCE(i.invoice_date, i.date::timestamp)) AS quarter,
  COUNT(*) AS invoice_count,
  SUM(COALESCE(i.amount, i.invoice_amount, 0)) AS total_spend,
  ROUND(SUM(COALESCE(i.amount, i.invoice_amount, 0)) * 0.04, 2) AS potential_savings_4pct
FROM invoices i
JOIN vendor_master vm ON i.vendor_id = vm.vendor_id
GROUP BY i.vendor_id, vm.vendor_name, vm.vendor_category,
         DATE_TRUNC('quarter', COALESCE(i.invoice_date, i.date::timestamp))
HAVING SUM(COALESCE(i.amount, i.invoice_amount, 0)) > 1000000;

-- 13. STORED PROCEDURE — Dynamic ingest
-- ============================================================
CREATE OR REPLACE FUNCTION ingest_data(
  target_table TEXT,
  data JSONB
) RETURNS TEXT AS
$$
DECLARE
  row_data JSONB;
  cols TEXT;
  vals TEXT;
  row_count INTEGER := 0;
BEGIN
  FOR row_data IN SELECT * FROM jsonb_array_elements(data) LOOP
    SELECT string_agg(key, ', '), string_agg(quote_literal(value), ', ')
    INTO cols, vals
    FROM jsonb_each_text(row_data);

    EXECUTE format('INSERT INTO %I (%s) VALUES (%s) ON CONFLICT DO NOTHING', target_table, cols, vals);
    row_count := row_count + 1;
  END LOOP;
  RETURN 'OK: ' || row_count || ' rows processed into ' || target_table;
END;
$$
LANGUAGE plpgsql;
